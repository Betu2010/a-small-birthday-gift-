  <canvas id="bouquetCanvas" width="800" height="800"></canvas>
  <p class="hint">Click anywhere on the canvas to replay the animation</p>


  <script>
    const canvas = document.getElementById("bouquetCanvas");
    const ctx = canvas.getContext("2d");


    // Colors
    const BG_COLOR = "#0c0c12";
    const GREEN = "#2e8b57";
    const STEM_GREEN = "#22703f";
    const DARK_GREEN = "#14502d";
    const PINK = "#eb578a";
    const RED = "#dc283c";
    const WHITE = "#f5f5fa";
    const YELLOW = "#ffd700";
    const ORANGE = "#ff8c00";
    const BROWN = "#654321";


    // Stem Bezier Data
    const stems = [
      // Left tall tulip
      { start: [400, 750], ctrl1: [320, 600], ctrl2: [220, 450], end: [260, 320], flower: "pink_tulip", delay: 0 },
      // Central sunflower
      { start: [400, 750], ctrl1: [410, 580], ctrl2: [390, 400], end: [400, 240], flower: "sunflower", delay: 20 },
      // Right daisy
      { start: [400, 750], ctrl1: [450, 600], ctrl2: [550, 480], end: [580, 340], flower: "white_daisy", delay: 40 },
      // Left lower red tulip
      { start: [400, 750], ctrl1: [350, 620], ctrl2: [300, 500], end: [340, 420], flower: "red_tulip", delay: 60 },
      // Right pink daisy
      { start: [400, 750], ctrl1: [430, 620], ctrl2: [480, 520], end: [500, 450], flower: "pink_daisy", delay: 80 },
      // Outer right tulip
      { start: [400, 750], ctrl1: [480, 580], ctrl2: [620, 450], end: [660, 300], flower: "red_tulip", delay: 100 }
    ];


    let frame = 0;
    const growthSpeed = 0.015;


    // Helper: Calculate Cubic Bezier Point at time t (0 to 1)
    function getBezierPoint(p0, p1, p2, p3, t) {
      const cx = 3 * (p1[0] - p0[0]);
      const bx = 3 * (p2[0] - p1[0]) - cx;
      const ax = p3[0] - p0[0] - cx - bx;


      const cy = 3 * (p1[1] - p0[1]);
      const by = 3 * (p2[1] - p1[1]) - cy;
      const ay = p3[1] - p0[1] - cy - by;


      const x = ax * Math.pow(t, 3) + bx * Math.pow(t, 2) + cx * t + p0[0];
      const y = ay * Math.pow(t, 3) + by * Math.pow(t, 2) + cy * t + p0[1];


      return { x, y };
    }


    // Flower Renderers
    function drawTulip(x, y, size, progress, color) {
      const s = size * progress;
      ctx.save();
      ctx.fillStyle = color;
      
      // Main Petal
      ctx.beginPath();
      ctx.ellipse(x, y - s / 2, s / 2, s * 0.6, 0, 0, Math.PI * 2);
      ctx.fill();


      // Outer Highlights
      ctx.strokeStyle = "rgba(255, 255, 255, 0.4)";
      ctx.lineWidth = 2;
      ctx.beginPath();
      ctx.ellipse(x - s / 4, y - s / 2, s / 3, s * 0.5, -0.2, 0, Math.PI * 2);
      ctx.stroke();


      ctx.restore();
    }


    function drawDaisy(x, y, size, progress, petalColor, centerColor) {
      const numPetals = 8;
      const radius = (size / 2) * progress;


      ctx.save();
      ctx.fillStyle = petalColor;


      // Petals
      for (let i = 0; i < numPetals; i++) {
        const angle = ((Math.PI * 2) / numPetals) * i;
        const px = x + Math.cos(angle) * (radius * 0.7);
        const py = y + Math.sin(angle) * (radius * 0.7);


        ctx.beginPath();
        ctx.arc(px, py, radius * 0.45, 0, Math.PI * 2);
        ctx.fill();
      }


      // Center
      ctx.fillStyle = centerColor;
      ctx.beginPath();
      ctx.arc(x, y, radius * 0.35, 0, Math.PI * 2);
      ctx.fill();


      ctx.restore();
    }


    function drawSunflower(x, y, size, progress) {
      const numPetals = 16;
      const radius = (size / 2) * progress;


      ctx.save();
      ctx.fillStyle = YELLOW;


      // Outer Petals
      for (let i = 0; i < numPetals; i++) {
        const angle = ((Math.PI * 2) / numPetals) * i;
        const px = x + Math.cos(angle) * (radius * 0.8);
        const py = y + Math.sin(angle) * (radius * 0.8);


        ctx.beginPath();
        ctx.arc(px, py, radius * 0.3, 0, Math.PI * 2);
        ctx.fill();
      }


      // Brown Center
      ctx.fillStyle = BROWN;
      ctx.beginPath();
      ctx.arc(x, y, radius * 0.55, 0, Math.PI * 2);
      ctx.fill();


      // Ring Accent
      ctx.strokeStyle = ORANGE;
      ctx.lineWidth = 3;
      ctx.stroke();


      ctx.restore();
    }


    function drawLeaf(x, y, length, angleDeg, progress) {
      if (progress <= 0) return;
      const l = length * progress;
      const rad = (angleDeg * Math.PI) / 180;


      const endX = x + l * Math.cos(rad);
      const endY = y + l * Math.sin(rad);
      const ctrlX = x + (l / 2) * Math.cos(rad - 0.5);
      const ctrlY = y + (l / 2) * Math.sin(rad - 0.5);


      ctx.save();
      ctx.strokeStyle = STEM_GREEN;
      ctx.lineWidth = 4;
      ctx.beginPath();
      ctx.moveTo(x, y);
      ctx.quadraticCurveTo(ctrlX, ctrlY, endX, endY);
      ctx.stroke();
      ctx.restore();
    }


    // Main Render Loop
    function animate() {
      frame++;
      ctx.clearRect(0, 0, canvas.width, canvas.height);


      // Background Leaf Base
      ctx.strokeStyle = DARK_GREEN;
      ctx.lineWidth = 3;
      for (let i = -5; i <= 5; i++) {
        const lx = 400 + i * 15;
        ctx.beginPath();
        ctx.arc(lx, 750, 100, Math.PI * 1.1, Math.PI * 1.9);
        ctx.stroke();
      }


      // Animate Stems and Flowers
      stems.forEach((stem) => {
        if (frame < stem.delay) return;


        const t = Math.min(1.0, (frame - stem.delay) * growthSpeed);


        // 1. Draw Growing Stem
        ctx.strokeStyle = GREEN;
        ctx.lineWidth = 5;
        ctx.lineCap = "round";
        ctx.beginPath();


        const steps = Math.floor(t * 50) + 1;
        for (let i = 0; i <= steps; i++) {
          const stepT = i / 50;
          const pt = getBezierPoint(stem.start, stem.ctrl1, stem.ctrl2, stem.end, stepT);
          if (i === 0) ctx.moveTo(pt.x, pt.y);
          else ctx.lineTo(pt.x, pt.y);
        }
        ctx.stroke();


        // 2. Leaves Growing
        if (t > 0.4) {
          const leafProgress = Math.min(1.0, (t - 0.4) / 0.6);
          const midPt = getBezierPoint(stem.start, stem.ctrl1, stem.ctrl2, stem.end, 0.45);
          drawLeaf(midPt.x, midPt.y, 40, -30, leafProgress);
          drawLeaf(midPt.x, midPt.y, 40, -150, leafProgress);
        }


        // 3. Flower Blooming
        if (t >= 0.8) {
          const bloomProgress = Math.min(1.0, (t - 0.8) / 0.2);
          const [endX, endY] = stem.end;


          switch (stem.flower) {
            case "pink_tulip":
              drawTulip(endX, endY, 50, bloomProgress, PINK);
              break;
            case "red_tulip":
              drawTulip(endX, endY, 45, bloomProgress, RED);
              break;
            case "white_daisy":
              drawDaisy(endX, endY, 60, bloomProgress, WHITE, YELLOW);
              break;
            case "pink_daisy":
              drawDaisy(endX, endY, 55, bloomProgress, PINK, YELLOW);
              break;
            case "sunflower":
              drawSunflower(endX, endY, 80, bloomProgress);
              break;
          }
        }
      });


      requestAnimationFrame(animate);
    }


    // Replay on Canvas Click
    canvas.addEventListener("click", () => {
      frame = 0;
    });


    // Start Animation
    animate();
  </script>
